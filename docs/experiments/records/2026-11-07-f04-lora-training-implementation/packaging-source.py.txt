from pathlib import Path
import json,hashlib,gzip,subprocess,time
R=Path('/mnt/nvme/kairyu/wsp/gemma4-coding-20261104')
B=R/'worktrees/kairyu-bench-weight-tools'
O=R/'analysis/fresh-purpose-20261107-lora-training-cpu-validation-v1'
N=R/'analysis/fresh-purpose-20261107-f04-mbpp-lora-training-v1'
D=B/'docs/experiments/records/2026-11-07-f04-lora-training-implementation'
assert not D.exists()
D.mkdir()
sha=lambda data:hashlib.sha256(data).hexdigest()
pins=json.loads((O/'final-source.json').read_text())['source_sha256']
for name,digest in pins.items():assert sha((B/name).read_bytes())==digest
inventory=[]
def store(path,name):
 raw=path.read_bytes();target=D/name
 if target.suffix=='.py':target=target.with_name(target.name+'.txt')
 compress=len(raw)>65536 or target.suffix in ('.patch','.diff')
 if compress:target=target.with_name(target.name+'.gz')
 target.parent.mkdir(parents=True,exist_ok=True)
 data=gzip.compress(raw,mtime=0) if compress else raw
 with target.open('xb') as f:f.write(data)
 assert (gzip.decompress(target.read_bytes()) if compress else target.read_bytes())==raw
 inventory.append({'source':str(path),'stored':str(target.relative_to(D)),'raw_sha256':sha(raw),'stored_sha256':sha(data),'raw_bytes':len(raw),'stored_bytes':len(data)})
for path in sorted(O.iterdir()):
 if path.is_file():store(path,'cpu/'+path.name)
for path in sorted((N/'merge-preparation-v1').iterdir()):
 if path.is_file():store(path,'merge-preparation/'+path.name)
for name in pins:store(B/name,'source/'+name)
diff=subprocess.check_output(['git','diff','--',*pins],cwd=B)
diffpath=O/'source-diff.patch'
with diffpath.open('xb') as f:f.write(diff)
store(diffpath,'source-diff.patch')
store(Path(__file__),'packaging-source.py.txt')
with (D/'packaging.json').open('x') as f:
 json.dump({'epoch':time.time(),'files':inventory,'source_sha256':pins,'source_unchanged_during_packaging':True,'GPU_or_model_operations':0,'note':'Source diff uses git tracked changes; newly added source/test are separately retained in full. Running GPU owner/training outputs are excluded.'},f,indent=2);f.write('\n')
files=sorted(x for x in D.rglob('*') if x.is_file())
(D/'SHA256SUMS').write_text(''.join(sha(x.read_bytes())+'  '+str(x.relative_to(D))+'\n' for x in files))
print(json.dumps({'records':str(D),'files':len(files)+1,'manifest_sha256':sha((D/'SHA256SUMS').read_bytes())}))
