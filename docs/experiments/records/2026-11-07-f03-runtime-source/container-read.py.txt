import importlib.metadata,json,hashlib
from pathlib import Path
rels=['transformers/models/gemma4/modeling_gemma4.py', 'vllm/model_executor/models/gemma4.py', 'vllm/model_executor/layers/layernorm.py', 'vllm/model_executor/models/utils.py', 'vllm/model_executor/model_loader/weight_utils.py', 'vllm/_custom_ops.py', 'vllm/model_executor/custom_op.py']
out={}
for rel in rels:
 d=importlib.metadata.distribution(rel.split('/')[0]);p=Path(d.locate_file(rel))
 raw=p.read_bytes();out[rel]=dict(path=str(p),version=d.version,sha256=hashlib.sha256(raw).hexdigest(),size=len(raw),text=raw.decode())
print(json.dumps(out))
