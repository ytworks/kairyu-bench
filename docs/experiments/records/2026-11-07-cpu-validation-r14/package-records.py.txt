from pathlib import Path
import ast,difflib,gzip,hashlib,json,re,shutil,time
R=Path('/mnt/nvme/kairyu/wsp/gemma4-coding-20261104');B=R/'worktrees/kairyu-bench-weight-tools'
P=R/'analysis/fresh-purpose-20261107-cpu-validation-r14';D=B/'docs/experiments/records/2026-11-07-cpu-validation-r14'
D.mkdir(exist_ok=False)
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
read=lambda p:json.loads(Path(p).read_text())
original=read(P/'completion.json');fix=read(P/'ruff-after-format.json')
assert original['success'] is False and original['full_suite_repetitions']==1
assert read(P/'sources-before.json')==read(P/'sources-after.json')
assert len(read(P/'sources-before.json'))==622
for row in original['results']:
 for stream in ('stdout','stderr'):
  k=stream+'_sha256'
  if k in row: assert sha(P/(row['name']+'.'+stream))==row[k]
stderr=(P/'unittest.stderr').read_text()
total,seconds=re.search(r'Ran (\d+) tests in ([\d.]+)s',stderr).groups()
skipped=int(re.search(r'OK \(skipped=(\d+)\)',stderr).group(1))
assert (int(total),skipped)==(592,60)
driver=B/'docs/experiments/scripts/fresh-purpose-cpu-validation-r14-20261107.py'
before=(P/'executed-driver-source.py.txt').read_bytes();after=driver.read_bytes()
assert hashlib.sha256(before).hexdigest()==fix['before_sha256']
assert hashlib.sha256(after).hexdigest()==fix['after_sha256']
ast_same=ast.dump(ast.parse(before),include_attributes=False)==ast.dump(ast.parse(after),include_attributes=False)
assert ast_same and fix['returncode']==0 and fix['unittest_repeated'] is False
archive={}
for p in sorted(P.iterdir()):
 if not p.is_file():continue
 data=p.read_bytes()
 destination=D/(p.name+'.gz' if len(data)>10000 else p.name)
 packed=gzip.compress(data,mtime=0) if len(data)>10000 else data
 destination.write_bytes(packed)
 assert (gzip.decompress(packed) if len(data)>10000 else packed)==data
 archive[p.name]=dict(source_sha256=hashlib.sha256(data).hexdigest(),source_bytes=len(data),stored=destination.name,stored_sha256=sha(destination),gzip=len(data)>10000)
(D/'formatted-driver-source.py.txt').write_bytes(after)
patch=''.join(difflib.unified_diff(before.decode().splitlines(True),after.decode().splitlines(True),fromfile='executed-driver-source.py.txt',tofile='formatted-driver-source.py.txt')).encode()
(D/'driver-format-only.diff.gz').write_bytes(gzip.compress(patch,mtime=0))
proof=dict(before_sha256=fix['before_sha256'],after_sha256=fix['after_sha256'],ast_identical_recomputed=ast_same,
           check='ast.dump(ast.parse(source), include_attributes=False)',executed_tests_source='original driver',
           no_tests_or_model_runs_during_packaging=True)
(D/'independent-format-ast-proof.json').write_text(json.dumps(proof,indent=2)+'\n')
summary=dict(status='passed_after_format_only_ruff_correction',initial_completion_success=False,
 initial_completion_sha256=sha(P/'completion.json'),initial_failed_check='ruff',initial_ruff_errors=5,
 tests=dict(total=int(total),passed=int(total)-skipped,skipped=skipped,failed=0,
            reported_seconds=float(seconds),wall_seconds=original['results'][0]['elapsed_seconds'],full_suite_repetitions=1),
 initial_all_checks_seconds=original['elapsed_seconds'],ruff_only_repeat_seconds=fix['seconds'],
 checks=dict(unittest='passed',ruff='passed_after_driver_format',mypy='passed_74_source_files',data='passed',tracked_shell='passed_31'),
 source_inventory_count=622,source_inventory_before_after_equal=True,
 pinned_experiment_sources_unchanged=original['explicit_pins_unchanged'],format_only_driver_changed_after_full_suite=proof,
 rerun=dict(unittest=0,mypy=0,data=0,shell=0,ruff=1),model_GPU_service_operations=0,
 archive=archive)
(D/'corrected-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
(P/'packaged-corrected-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
doc=B/'docs/experiments/2026-11-07-cpu-validation-r14.md'
