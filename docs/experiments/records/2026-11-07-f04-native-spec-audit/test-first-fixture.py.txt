class NativeCoupledTests(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.root = Path(self.tmp.name)
        self.norm = "model.language_model.norm.weight"
        self.scalar = "model.language_model.layers.0.layer_scalar"
        self.matrix = "model.language_model.layers.0.mlp.down_proj.weight"
        model = self.root / "model"
        weights(model / "model.safetensors", {
            self.norm: ("BF16", (2,), bytes.fromhex("803f0040")),
            self.scalar: ("BF16", (), bytes.fromhex("0080")),
            self.matrix: ("BF16", (1, 1), bytes.fromhex("803f")),
        })
        self.checkpoint = Checkpoint(model)
        identity = self.checkpoint.identity()
        attestation = self.root / "attestation.json"
        attestation.write_text(json.dumps({"checkpoint": identity}))
        self.cases = self.root / "cases.json"
        self.cases.write_text(json.dumps({"science": [[{
            "id": "authored", "prompt": "p", "correct": "good", "foil": "bad"}]]}))
        self.case_sha = self.sha(self.cases)
        coverage = {"native_tensors": 2, "virtual_tensors": 0, "native_coordinates": 3,
                    "virtual_coordinates": 0, "tensors": [
                        {"name": self.norm, "shape": [2], "dtype": "torch.bfloat16", "kind": "native"},
                        {"name": self.scalar, "shape": [], "dtype": "torch.bfloat16", "kind": "native"}]}
        self.directories = {}
        for objective in COUPLED.OBJECTIVES:
            path = self.root / objective
            path.mkdir()
            self.directories[objective] = str(path)
            values = np.array([1, 2], dtype=np.float32)
            np.savez(path / "source-native-values.npz", **{self.norm: values, self.scalar: np.array(-0., dtype=np.float32)})
            gradients = [1., 0.] if objective == "correct-logprob" else [-1., 1.]
            np.savez(path / "gradient-science-0.npz", **{self.norm: np.array(gradients), self.scalar: np.array(0.)})
            field, value = ("correct_logprobs", "correct_logprob") if objective == "correct-logprob" else ("margins", "margin")
            summary = {"status": "completed", "candidate_selection": False, "source_values_unchanged": True,
                       "mode": "chat-public-answer", "objective_kind": objective, "answer_batching": "serial",
                       "runtime": {"torch": "fixture", "transformers": "fixture", "weight_dtype": "BF16",
                                   "attention": "eager", "gradients": "FP32 accumulation",
                                   "max_cases_per_microbatch": 1,
                                   "gpu_peak_allocated_GiB": 1 if objective == "margin" else 2},
                       "folds": [{"purpose": "science", "fold": 0, "examples": 1, "file": "gradient-science-0.npz",
                                  field: [{"id": "authored", value: -1. if objective == "correct-logprob" else 0.}]}]}
            provenance = {"mode": "chat-public-answer", "objective_kind": objective, "answer_batching": "serial",
                          "virtual_unit_gains": False, "max_cases_per_microbatch": 1, "threads": 4, "seed": 0,
                          "checkpoint": self.checkpoint.identity(False),
                          "attestation_receipt_sha256": self.sha(attestation),
                          "input_sha256": {str(self.cases): self.case_sha}, "input_stat": {},
                          "calibration_scope": "authored fixture"}
            frame = {"schema_version": 2, "frame_protocol": "gemma4-public-answer-v2-forced-empty-thought",
                     "forced_empty_thought": True, "frames": {"science": [[{"id": "authored"}]]}}
            for name, data in [("summary.json", summary), ("provenance.json", provenance),
                               ("coverage.json", coverage), ("frame-evidence.json", frame)]:
                (path / name).write_text(json.dumps(data))
        self.plan = {"schema_version": 1, "kind": "native_coupled_objectives_v1",
                     "checkpoint": {"path": str(model), "tree_sha256": identity["sha256"],
                                    "attestation": {"path": str(attestation), "sha256": self.sha(attestation)}},
                     "cases": {"path": str(self.cases), "sha256": self.case_sha},
                     "collections": self.directories,
                     "candidates": [{"id": "scope-a", "constraints": [
                         {"objective": o, "purpose": "science", "fold": 0, "minimum_delta": 1 / 128}
                         for o in COUPLED.OBJECTIVES], "cost_metric": {"type": "euclidean"},
                         "rounding": {"relative_radius": .025, "absolute_radius": 1e-6, "rounds": 2}}]}
        self.plan_path = self.root / "plan.json"
        self.plan_path.write_text(json.dumps(self.plan))

    def mixed_fixture(self):
        tensors = {
            self.norm: ("BF16", (2,), bytes.fromhex("803f0040")),
            self.scalar: ("BF16", (), bytes.fromhex("0080")),
        }
        for kind, shape in (("gate", (16, 32)), ("up", (16, 32)), ("down", (32, 16)),
                            ("o", (32, 32)), ("q", (32, 32))):
            area = "mlp" if kind in {"gate", "up", "down"} else "self_attn"
            name = f"model.language_model.layers.0.{area}.{kind}_proj.weight"
            tensors[name] = ("BF16", shape, bytes.fromhex("803f") * int(np.prod(shape)))
        tensors["model.language_model.layers.1.self_attn.o_proj.weight"] = (
            "BF16", (32, 32), bytes.fromhex("803f") * 1024)
        weights(self.checkpoint.path / "model.safetensors", tensors)
        self.checkpoint = Checkpoint(self.checkpoint.path)
        identity = self.checkpoint.identity()
        attestation = self.root / "attestation.json"
        attestation.write_text(json.dumps({"checkpoint": identity}))
        self.plan["schema_version"] = 2
        self.plan["kind"] = "native_matrix_coupled_objectives_v2"
        self.plan["checkpoint"].update(tree_sha256=identity["sha256"],
                                       attestation={"path": str(attestation), "sha256": self.sha(attestation)})
        self.plan["baseline_tolerance"] = 1e-6
        self.plan["collections"] = {"native": dict(self.directories), "matrix": {}}
        self.plan["candidates"][0]["cost_metric"] = {
            "type": "relative", "floor": 1e-6, "block_scales": {"native": 1., "matrix": 1.}}
        coherent = COUPLED.matrix_recipe.mlp_triplet_samples(self.checkpoint.tensors, 0)
        entries, coords, source = [], [], []
        for name, tensor in sorted(self.checkpoint.tensors.items()):
            match = COUPLED.matrix_recipe.PATTERN.fullmatch(name)
            if not match:
                continue
            rc, strata = coherent.get(name, (np.array([[0, 0]], dtype=np.int64), {}))
            entries.append({"tensor": name, "shape": list(tensor.shape), "count": len(rc),
                            "start": len(source), "family": match[3], "layer": int(match[1]), "strata": strata})
            coords.extend(rc)
            source.extend([1.] * len(rc))
        coverage = {"entries": entries, "sampled_matrices": len(entries),
                    "sampled_coordinates": len(source), "sampling_mode": "mlp-triplets"}
        for objective, directory in self.directories.items():
            path = Path(directory)
            provenance = json.loads((path / "provenance.json").read_text())
            provenance.update(checkpoint=self.checkpoint.identity(False), attestation_receipt_sha256=self.sha(attestation))
            (path / "provenance.json").write_text(json.dumps(provenance))
            if objective == "correct-logprob":
                np.savez(path / "gradient-science-0.npz", **{self.norm: np.zeros(2), self.scalar: np.array(0.)})
            target = self.root / ("matrix-" + objective)
            target.mkdir()
            self.plan["collections"]["matrix"][objective] = str(target)
            np.savez(target / "sample-plan.npz", coordinates=np.asarray(coords), source_values=np.asarray(source, dtype=np.float32))
            gradient = np.zeros(len(source))
            if objective == "correct-logprob":
                for entry in entries:
                    if entry["family"] == "gate":
                        gradient[entry["start"]] = 1.
                    if entry["family"] == "q":
                        gradient[entry["start"]] = 1e6  # Diagnostic-only temptation must not enter the solve.
            np.save(target / "gradient.npy", gradient)
            summary = json.loads((path / "summary.json").read_text())
            summary["folds"][0]["file"] = "gradient.npy"
            summary["runtime"].pop("gradients")
            summary["runtime"].update(gradient_accumulation="FP32 factors, FP64 sum", gradient_checkpointing=False)
            matrix_provenance = dict(provenance, sampling_mode="mlp-triplets")
            matrix_provenance.pop("virtual_unit_gains")
            for name, document in (("summary.json", summary), ("provenance.json", matrix_provenance),
                                   ("coverage.json", coverage),
                                   ("frame-evidence.json", json.loads((path / "frame-evidence.json").read_text()))):
                (target / name).write_text(json.dumps(document))
        self.plan_path.write_text(json.dumps(self.plan))
        return entries

    @staticmethod
    def sha(path):
        return hashlib.sha256(path.read_bytes()).hexdigest()

    def test_two_objectives_produce_exact_bf16_spec_without_touching_source(self):
        before = {p: self.sha(p) for p in self.checkpoint.files}
        _, _, _, labels, _ = COUPLED.prepare(self.plan_path)
        self.assertEqual(labels, [("science", 0)])
        with unittest.mock.patch.object(COUPLED.joint, "read_collection", side_effect=AssertionError("preview read gradient")):
            COUPLED.main(["--plan", str(self.plan_path), "--output", str(self.root / "preview")])
        self.assertFalse((self.root / "preview").exists())
        output = self.root / "proposal"
        report = COUPLED.solve(self.plan_path, output, write_spec=True)
        actual = report["methods"][0]
        self.assertTrue(actual["predicted_constraints_satisfied"])
        self.assertEqual(actual["changed_elements"], 2)
        spec = json.loads((output / "scope-a.json").read_text())
        self.assertEqual(spec["edits"], [{"tensor": self.norm, "indices": [0, 1], "set_values": [1.0078125, 2.015625]}])
        self.assertEqual(actual["prediction"], [1 / 128, 1 / 128])
        self.assertEqual({p: self.sha(p) for p in self.checkpoint.files}, before)
        self.assertEqual(self.checkpoint.read_raw(self.scalar), bytes.fromhex("0080"))
        # The report must reach completion with both a numeric layer and the
        # final norm's global scope, rather than crashing after all reconstruction.
        audit_plan = copy.deepcopy(self.plan)
        audit_plan["native_tensors"] = [self.scalar, self.norm]
        dependency = self.root / "audit-dependency.json"
        dependency.write_text(json.dumps({"case_rows": [
            {"id": "authored", "correct_logprob": -1., "margin": 0.}]}))
        parent = self.root / "audit-parent.json"
        parent.write_text(json.dumps({"dependency": {
            "path": str(dependency), "sha256": self.sha(dependency)}}))
        audit_plan["parent_state"] = {"path": str(parent), "sha256": self.sha(parent)}
        audit_path = self.root / "proposal-plan.json"
        audit_path.write_text(json.dumps(audit_plan))
        (self.root / "solver-execution.json").write_text(json.dumps({"status": "CPU fixture"}))
        audit = load("native_spec_audit").audit(self.root, self.sha(audit_path))
        self.assertEqual(audit["status"], "existing_native_spec_independently_reconstructed")
        self.assertEqual(audit["eligible_layers"], ["0", "global"])
        self.assertEqual(audit["eligible_scope"]["changed"], actual["changed_elements"])
        self.assertEqual(audit["met_constraints"], 2)
        with self.assertRaisesRegex(ValueError, "fresh output"):
            COUPLED.solve(self.plan_path, output, write_spec=True)
        entries = self.mixed_fixture()
        before = {p: self.sha(p) for p in self.checkpoint.files}
        output = self.root / "mixed-proposal"
        report = COUPLED.solve(self.plan_path, output, write_spec=True)
        actual = report["methods"][0]
        spec_path = output / "scope-a.json"
        spec = json.loads(spec_path.read_text())
        self.assertTrue(spec["matrix_batch"])
        self.assertTrue(any("indices" in e for e in spec["edits"]))
        self.assertTrue(any("coordinates" in e for e in spec["edits"]))
        self.assertFalse(any(".q_proj." in e["tensor"] for e in spec["edits"]))
        self.assertTrue(actual["predicted_constraints_satisfied"])
        self.assertEqual(actual["column_scope"]["excluded_qkv_columns"], 1)
        self.assertTrue(any(row["changed"] == 0 for row in actual["realized_matrix_support"]))
        candidate, manifest = self.root / "saved", self.root / "saved.json"
        editor.create(self.checkpoint.path, candidate, spec_path, manifest, apply=True, reserve_bytes=0)
        editor.verify(manifest)
        saved = Checkpoint(candidate)
        qname = next(e["tensor"] for e in entries if e["family"] == "q")
        self.assertEqual(saved.read_raw(qname), self.checkpoint.read_raw(qname))
        self.assertEqual(saved.read_raw(self.scalar), bytes.fromhex("0080"))
        self.assertEqual({p: self.sha(p) for p in self.checkpoint.files}, before)
        # A separate completed purpose is copied into one ordinary adapter input
        # without rerunning, averaging or altering the first purpose's gradients.
        with unittest.mock.patch.dict(sys.modules, {"propose_native_objectives": COUPLED}):
            composer = load("compose_purpose_collections")
        new_cases = self.root / "helper-cases.json"
        new_cases.write_text(json.dumps({"helper": [[{
            "id": "helper-authored", "prompt": "hp", "correct": "yes", "foil": "no"}]]}))
        full_cases = self.root / "combined-cases.json"
        combined = json.loads(self.cases.read_text())
        combined.update(json.loads(new_cases.read_text()))
        full_cases.write_text(json.dumps(combined))
        parent = self.root / "parent-state.json"
        parent.write_text(json.dumps({"accepted": "fixture"}))
        new_directories = {family: {} for family in ("native", "matrix")}
        for family, objectives in self.plan["collections"].items():
            for objective, source in objectives.items():
                target = self.root / ("helper-" + family + "-" + objective)
                shutil.copytree(source, target)
                new_directories[family][objective] = str(target)
                summary = json.loads((target / "summary.json").read_text())
                row = summary["folds"][0]
                row["purpose"] = "helper"
                field = "correct_logprobs" if objective == "correct-logprob" else "margins"
                row[field][0]["id"] = "helper-authored"
                (target / "summary.json").write_text(json.dumps(summary))
                frame = json.loads((target / "frame-evidence.json").read_text())
                frame["frames"] = {"helper": [[{"id": "helper-authored"}]]}
                (target / "frame-evidence.json").write_text(json.dumps(frame))
                provenance = json.loads((target / "provenance.json").read_text())
                provenance["input_sha256"] = {str(new_cases): self.sha(new_cases)}
                (target / "provenance.json").write_text(json.dumps(provenance))
        def collection_receipt(name, directories):
            completion = self.root / (name + "-completion.json")
            if not completion.exists():
                completion.write_text(json.dumps({"status": "completed"}))
            outputs = {}
            for family, objectives in directories.items():
                for objective, directory in objectives.items():
                    paths = list(Path(directory).iterdir())
                    outputs[family + "-" + objective] = {
                        "path": directory, "files": {p.name: self.sha(p) for p in paths},
                        "stat": {str(p): COUPLED.stamp(p) for p in paths}}
            receipt = self.root / (name + "-receipt.json")
            receipt.write_text(json.dumps({
                "all_stat_stable": True,
                "completion": {"path": str(completion), "sha256": self.sha(completion)},
                "source_parent": {"path": str(parent), "sha256": self.sha(parent),
                                  "tree": self.plan["checkpoint"]["tree_sha256"]},
                "outputs": outputs}))
            return {"path": str(receipt), "sha256": self.sha(receipt)}
        # Legacy inputs above remain accepted. An opt-in backend/frame policy
        # must survive composition, and cannot differ within or between segments.
        observation_policy = {
            "max_frame_tokens": 1024,
            "attention_dispatch": {"": "eager", "text_config": "sdpa",
                                   "vision_config": "eager", "audio_config": "eager"},
            "cuda_memory_cap_gib": 88.0, "answer_batching": "serial",
            "max_cases_per_microbatch": 1, "execution_order": "original"}
        def set_observation_policy(directories, policy):
            for objectives in directories.values():
                for directory in objectives.values():
                    folder = Path(directory)
                    summary = json.loads((folder / "summary.json").read_text())
                    provenance = json.loads((folder / "provenance.json").read_text())
                    summary["runtime"]["observation_policy"] = copy.deepcopy(policy)
                    summary["runtime"]["attention"] = policy["attention_dispatch"]["text_config"]
                    provenance["observation_policy"] = copy.deepcopy(policy)
                    (folder / "summary.json").write_text(json.dumps(summary))
                    (folder / "provenance.json").write_text(json.dumps(provenance))
        set_observation_policy(self.plan["collections"], observation_policy)
        set_observation_policy(new_directories, observation_policy)
        compose_plan = {
            "schema_version": 1, "kind": "disjoint_purpose_collections_v1",
            "checkpoint": self.plan["checkpoint"], "baseline_tolerance": 1e-6,
            "parent_state": {"path": str(parent), "sha256": self.sha(parent)},
            "cases": {"path": str(full_cases), "sha256": self.sha(full_cases)},
            "segments": [
                {"cases": self.plan["cases"], "receipt": collection_receipt("old", self.plan["collections"])},
                {"cases": {"path": str(new_cases), "sha256": self.sha(new_cases)},
                 "receipt": collection_receipt("new", new_directories)}]}
        compose_path = self.root / "compose-plan.json"
        compose_path.write_text(json.dumps(compose_plan))
        different_policy = dict(observation_policy, max_frame_tokens=2048)
        bad_summary = Path(new_directories["matrix"]["margin"]) / "summary.json"
        broken = json.loads(bad_summary.read_text())
        broken["runtime"]["observation_policy"] = different_policy
        bad_summary.write_text(json.dumps(broken))
        compose_plan["segments"][1]["receipt"] = collection_receipt("new", new_directories)
        compose_path.write_text(json.dumps(compose_plan))
        with self.assertRaisesRegex(ValueError, "observation policy/runtime"):
            composer.compose(compose_path, self.root / "mismatched-policy", write=True)
        self.assertFalse((self.root / "mismatched-policy").exists())
        set_observation_policy(new_directories, different_policy)
        compose_plan["segments"][1]["receipt"] = collection_receipt("new", new_directories)
        compose_path.write_text(json.dumps(compose_plan))
        with self.assertRaisesRegex(ValueError, "collector settings differ"):
            composer.compose(compose_path, self.root / "mixed-frame-limits", write=True)
        self.assertFalse((self.root / "mixed-frame-limits").exists())
        set_observation_policy(new_directories, observation_policy)
        compose_plan["segments"][1]["receipt"] = collection_receipt("new", new_directories)
        compose_path.write_text(json.dumps(compose_plan))
        composed = self.root / "composed"
        with unittest.mock.patch.object(COUPLED.joint, "quantized_proposal", side_effect=AssertionError("composer invoked solver")):
            preview = composer.compose(compose_path, composed)
            self.assertEqual(preview["folds"], 2)
            self.assertFalse(composed.exists())
            result = composer.compose(compose_path, composed, write=True)
        self.assertTrue(result["composition"]["not_a_single_model_run"])
        self.assertEqual(result["model_calls"], 0)
        for family, objectives in self.plan["collections"].items():
            for objective, source in objectives.items():
                original = Path(source)
                filename = json.loads((original / "summary.json").read_text())["folds"][0]["file"]
                self.assertEqual((composed / (family + "-" + objective) / ("segment-0-" + filename)).read_bytes(),
                                 (original / filename).read_bytes())
        joined_plan_path = composed / "adapter-input-template.json"
        _, joined_plan, joined_collections, joined_labels, _ = COUPLED.prepare(joined_plan_path)
        self.assertTrue(all(c["summary"]["runtime"]["observation_policy"] == observation_policy
                            and c["provenance"]["observation_policy"] == observation_policy
                            for c in joined_collections.values()))
        self.assertEqual(joined_labels, [("science", 0), ("helper", 0)])
        joined_plan["candidates"][0]["constraints"] = [
            {"objective": o, "purpose": p, "fold": f, "minimum_delta": 1 / 128}
            for o in COUPLED.OBJECTIVES for p, f in joined_labels]
        joined_plan["candidates"][0]["rounding"]["relative_radius"] = .025
        proposal_plan = self.root / "composed-proposal-plan.json"
        proposal_plan.write_text(json.dumps(joined_plan))
        proposal_output = self.root / "composed-proposal"
        COUPLED.solve(proposal_plan, proposal_output, write_spec=True)
        saved_path, manifest_path = self.root / "composed-saved", self.root / "composed-saved.json"
        editor.create(self.checkpoint.path, saved_path,
                      proposal_output / "composition-validation-only.json",
                      manifest_path, apply=True, reserve_bytes=0)
        editor.verify(manifest_path)
        self.assertEqual(Checkpoint(saved_path).read_raw(qname), self.checkpoint.read_raw(qname))
        self.assertEqual(Checkpoint(saved_path).read_raw(self.scalar), bytes.fromhex("0080"))
        selected = copy.deepcopy(compose_plan)
        selected["purpose_selection"] = {
            "selected_purposes": ["helper"], "omitted_reasons": {"science": "fixture selection"}}
        compose_path.write_text(json.dumps(selected))
        subset = self.root / "composed-subset"
        selected_result = composer.compose(compose_path, subset, write=True)
        self.assertEqual((selected_result["cases"], selected_result["source_folds"]), (1, 2))
        _, selected_plan, _, selected_labels, _ = COUPLED.prepare(subset / "adapter-input-template.json")
        self.assertEqual(selected_labels, [("helper", 0)])
        self.assertEqual(selected_plan["cases"]["path"], str(subset / "cases.json"))
        for family, objectives in new_directories.items():
            for objective, source in objectives.items():
                filename = json.loads((Path(source) / "summary.json").read_text())["folds"][0]["file"]
                self.assertEqual((subset / (family + "-" + objective) / ("segment-1-" + filename)).read_bytes(),
                                 (Path(source) / filename).read_bytes())
        compose_path.write_text(json.dumps(compose_plan))
        overlapping = copy.deepcopy(compose_plan)
        overlapping["segments"][1] = overlapping["segments"][0]
        compose_path.write_text(json.dumps(overlapping))
        with self.assertRaisesRegex(ValueError, "purposes must be disjoint"):
            composer.compose(compose_path, self.root / "overlap", write=True)
        self.assertFalse((self.root / "overlap").exists())
        # Numerical equality is insufficient for signed-zero source identity.
        altered = Path(new_directories["native"]["margin"]) / "source-native-values.npz"
        np.savez(altered, **{self.norm: np.array([1., 2.], dtype=np.float32),
                            self.scalar: np.array(0., dtype=np.float32)})
        compose_plan["segments"][1]["receipt"] = collection_receipt("new", new_directories)
        compose_path.write_text(json.dumps(compose_plan))
        with self.assertRaisesRegex(ValueError, "source array/sample bytes differ"):
            composer.compose(compose_path, self.root / "changed-source", write=True)
        self.assertFalse((self.root / "changed-source").exists())

        # A new, explicit matrix-only hypothesis keeps all native source bytes fixed.
        self.plan["eligible_blocks"] = ["matrix"]
        self.plan["candidates"][0]["id"] = "matrix-only"
        margin_dir = Path(self.plan["collections"]["matrix"]["margin"])
        gradient = np.load(margin_dir / "gradient.npy", allow_pickle=False)
        up = next(e for e in entries if e["family"] == "up")
        gradient[up["start"]] = 1.
        np.save(margin_dir / "gradient.npy", gradient)
        plan_path = self.root / "matrix-only-plan.json"
        plan_path.write_text(json.dumps(self.plan))
        output = self.root / "matrix-only-proposal"
        report = COUPLED.solve(plan_path, output, write_spec=True)
        actual = report["methods"][0]
        spec_path = output / "matrix-only.json"
        spec = json.loads(spec_path.read_text())
        self.assertTrue(actual["predicted_constraints_satisfied"])

        summary_path = path.parent / "summary.json"

        with self.assertRaisesRegex(ValueError, "cohort"):
            COUPLED.prepare(self.plan_path)
        self.assertFalse((self.root / "proposal").exists())
        # Fresh four-way data cannot accept stale whole-source evidence or permuted matrix columns.
        summary["folds"][0]["margins"][0]["id"] = "authored"
        summary_path.write_text(json.dumps(summary))
        self.mixed_fixture()

        self.assertFalse((self.root / "bad-mixed").exists())


class NativeDependencyGeometryTests(unittest.TestCase):
    """Protect zero-gradient unknowns and BF16 exponent-boundary asymmetry."""

    def test_group_contribution_preserves_opposition_and_zero_is_unknown(self):
        dependency = load("native_dependency")
        gradient = np.array([[1., 2.], [-1., 2.], [0., 0.]])
        whole = gradient @ gradient.T
        left = gradient[:, :1] @ gradient[:, :1].T
        right = gradient[:, 1:] @ gradient[:, 1:].T
        a = dependency.geometry(left, whole)
        b = dependency.geometry(right, whole)
        total = dependency.geometry(whole, whole)
        self.assertAlmostEqual(a["cosine"][0][1], -1.)
        self.assertAlmostEqual(b["cosine"][0][1], 1.)
        self.assertAlmostEqual(total["cosine"][0][1], .6)
        self.assertAlmostEqual(a["contribution_to_global_cosine"][0][1]
                               + b["contribution_to_global_cosine"][0][1], .6)
        self.assertIsNone(total["cosine"][0][2])
        self.assertIsNone(a["gradient_energy_fraction"][2])

    def test_bf16_spacing_handles_signed_zero_and_asymmetric_powers(self):
        dependency = load("native_dependency")
        up, down = dependency.bf16_gaps(np.array([1., -1., 0., -0.], dtype=np.float32))
        np.testing.assert_array_equal(up, [2**-7, 2**-8, 2**-133, 2**-133])
        np.testing.assert_array_equal(down, [2**-8, 2**-7, 2**-133, 2**-133])
        with self.assertRaisesRegex(ValueError, "exact BF16"):
            dependency.bf16_gaps(np.array([1.001], dtype=np.float32))


class NativeSpecAuditTests(unittest.TestCase):
    def test_realized_box_excess_and_zero_denominator_are_not_hidden(self):
        audit = load("native_spec_audit")
        values = np.array([1.25, -1.25, 0.], dtype=np.float32)
        target = audit.apply_edit(values, {"tensor": "fixture", "indices": [0, 1],
                                          "set_values": [1.265625, -1.265625]})
        row = audit.diff_stats(values, target, .01, 1e-6)
        self.assertEqual((row["positive_delta"], row["negative_delta"], row["unchanged"]), (1, 1, 1))
        self.assertEqual(row["exceeds_declared_continuous_box"], 2)
        self.assertAlmostEqual(row["maximum_actual_to_box_ratio"], 1.25)
        self.assertEqual(row["relative_delta_undefined_on_zero_source"], 1)
        zero = audit.diff_stats(np.array([0.]), np.array([0.]), .01, 1e-6)
        self.assertIsNone(zero["relative_delta_l2"])
        self.assertIsNone(zero["max_relative_delta_on_nonzero_source"])

    def test_sparse_spec_cannot_overwrite_duplicate_or_round_invalid_values(self):
        audit = load("native_spec_audit")
        base = np.array([1.], dtype=np.float32)
        with self.assertRaisesRegex(ValueError, "duplicate"):
            audit.apply_edit(base, {"tensor": "fixture", "indices": [0, 0], "set_values": [1., 2.]})
        with self.assertRaisesRegex(ValueError, "exact finite BF16"):
            audit.apply_edit(base, {"tensor": "fixture", "indices": [0], "set_values": [1.001]})
