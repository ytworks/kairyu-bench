"""Package the one completed CPU validation without rerunning tests."""
import collections
import gzip
import hashlib
import json
from pathlib import Path
import re
import time

R = Path("/mnt/nvme/kairyu/wsp/gemma4-coding-20261104")
B = R / "worktrees/kairyu-bench-weight-tools"
D = R / "analysis/fresh-purpose-20261107-cpu-validation-r11"
PR = B / "docs/experiments/records/2026-11-07-cpu-validation-r11"
PR.mkdir(exist_ok=False)
def digest(raw):
    return hashlib.sha256(raw).hexdigest()
completion=json.loads((D/"completion.json").read_text())
assert completion["success"] and completion["sources_unchanged"]
log=(D/"unittest.stderr").read_text()
match=re.search(r"Ran (\d+) tests in ([0-9.]+)s",log)
assert match
skip_match=re.search(r"OK \(skipped=(\d+)\)",log)
count,skipped=int(match[1]),int(skip_match[1])if skip_match else 0
assert re.search(r"\nOK(?: \(skipped=\d+\))?\s*$",log)
mypy=(D/"mypy.stdout").read_text()
mypy_match=re.search(r"Success: no issues found in (\d+) source files",mypy)
assert mypy_match
skip_reasons=collections.Counter(re.findall(r"\.\.\. skipped (.+)",log))
summary={
    "status":"all_requested_CPU_checks_passed",
    "unittest":{"count":count,"passed":count-skipped,"skipped":skipped,"reported_seconds":float(match[2]),"wall_seconds":completion["results"][0]["elapsed_seconds"],"skip_reasons":dict(skip_reasons)},
    "ruff_passed":True,"mypy_passed":True,"mypy_source_files":int(mypy_match[1]),
    "data_check_passed":True,"tracked_shell_files":31,"tracked_shell_syntax_passed":True,
    "sources_unchanged":True,"source_count":completion["source_count_after"],
    "explicit_train_dense_owner_pins_unchanged":True,"full_suite_repetitions":1,
    "elapsed_seconds":completion["elapsed_seconds"],"model_GPU_service_operations":0,
    "parallel_activity":"Root/observation agent GPU trial independent; this driver ran CPU checks only.",
    "shell_scope_note":"Parent explicitly requested tracked repository syntax. All tracked .sh checked individually. The already-recorded r10 ignored-cache traversal error was not repeated.",
    "PR33_v7_body_sha256":"58fe5525d2895ccd71a71efa5416774c4e63f8ceb51c6147eb0b82601d22157a",
}
(D/"summary.json").write_text(json.dumps(summary,indent=2)+"\n")
artifacts=[]
for source in [*sorted(p for p in D.iterdir()if p.is_file()),R/"analysis/fresh-purpose-20261107-cpu-validation-r11-run.py.txt",Path(__file__)]:
    raw=source.read_bytes()
    compressed=len(raw)>65536
    name=source.name+(".gz"if compressed else "")
    stored=gzip.compress(raw,mtime=0)if compressed else raw
    with(PR/name).open("xb")as stream:stream.write(stored)
    if compressed:assert gzip.decompress(stored)==raw
    artifacts.append({"name":name,"raw_sha256":digest(raw),"stored_sha256":digest(stored),"raw_bytes":len(raw),"gzip":compressed})
with(PR/"artifact-index.json").open("x")as stream:json.dump(artifacts,stream,indent=2);stream.write("\n")
files=sorted(p for p in PR.iterdir()if p.is_file())
with(PR/"SHA256SUMS").open("x")as stream:
    stream.write("".join(digest(p.read_bytes())+"  "+p.name+"\n"for p in files))

Full Ruff, mypy on {int(mypy_match[1])} source files, generated comparison-data verification and all31 tracked repository shell syntax checks passed. All320 monitored Python/config/tracked-shell sources matched before/after, including the explicit trainer §8bfa6f3f5568264417d52e1cf297ebf0f988458c0c075f329a189a0233a7a846§, dense merger §5c06f59a2734f94a2e50310392a7959e9b8f73f012f0cef646df13bf79dae84c§ and owner §3e7e9c58b68a679d87623e1507e6508adcc731c3da36d51dd390f9bffcdb2b07§. Concurrent Markdown/record additions were excluded from source continuity. Execution used the existing CPU venv, nice15, one-thread numerical libraries and the short owned NVMe TMPDIR.

The r10 cache-permission failure remains intact in its earlier record.
