selected = ['basic', 'advanced-v2', 'library-train-v1']

parts = {name: {} for name in partition_names}
mapping = []
changes = []
for purpose, fold, index, row in rows:
    matches = []
    for name, catalog in catalogs.items():
        if row['id'] not in catalog:
            continue
        other = catalog[row['id']]
        differing = sorted(k for k in set(row) | set(other) if row.get(k) != other.get(k))
        item = {'catalog': name, 'split': other['split'], 'id_match': True, 'prompt_exact': row['prompt'] == other['prompt'], 'full_row_exact': row == other, 'differing_fields': differing, 'catalog_prompt_sha256': digest(other['prompt'])}
        if row['prompt'] != other['prompt']:
            item['prompt_unified_diff_source_to_catalog'] = ''.join(difflib.unified_diff(row['prompt'].splitlines(True), other['prompt'].splitlines(True), fromfile='authored120', tofile=name))
        matches.append(item)
    eligible = [m['catalog'] for m in matches if m['catalog'] in selected and m['full_row_exact'] and m['split'] == 'train']
    assert len(eligible) <= 1
    partition = eligible[0] if eligible else 'uncataloged72'
    derived = copy.deepcopy(row)
    if 'split' not in derived:
        derived['split'] = 'train'
        changes.append({'id': row['id'], 'operation': 'add_missing_split_train', 'source_purpose': purpose, 'source_fold': fold, 'source_index': index})
    assert derived['split'] == 'train'
    assert {k: derived[k] for k in row} == row
    if set(derived) - set(row):
        assert set(derived) - set(row) == {'split'}
    parts[partition].setdefault(purpose, [[] for _ in groups[purpose]])[fold].append(derived)
    mapping.append({'id': row['id'], 'purpose': purpose, 'fold': fold, 'source_index': index, 'original_split': row.get('split'), 'prompt': row['prompt'], 'prompt_sha256': digest(row['prompt']), 'correct_sha256': digest(row['correct']), 'foil_sha256': digest(row['foil']), 'original_row_canonical_sha256': digest(canonical(row)), 'partition': partition, 'catalog_matches': matches, 'existing_5catalog_executor_eligible': bool(eligible), 'known_calibration': True, 'audit_counterexample': issues.get(row['id']), 'generation_status': 'not_run', 'execution_status': 'not_run' if eligible else 'unscored_outside_5catalogs'})
