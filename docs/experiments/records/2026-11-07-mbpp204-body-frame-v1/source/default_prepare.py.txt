import hashlib
import json
FRAME_PROTOCOL = 'gemma4-sft-standard-on-answer-eos-v1'

def prepare_training_frames(tokenizer, rows):
    """Exact official ON dialog; no invented reasoning, padding or truncation."""
    frames, evidence = ([], [])
    for row in rows:
        user = {'role': 'user', 'content': row['text']}
        prefix = tokenizer.apply_chat_template([user], tokenize=False, enable_thinking=True, add_generation_prompt=True)
        full = tokenizer.apply_chat_template([user, {'role': 'assistant', 'content': row['code']}], tokenize=False, enable_thinking=True, add_generation_prompt=False)
        pids = tokenizer.encode(prefix, add_special_tokens=False)
        ids = tokenizer.encode(full, add_special_tokens=False)
        if not pids or ids[:len(pids)] != pids or len(ids) <= len(pids):
            raise ValueError('official prefix/answer token boundary mismatch')
        if '<|channel>thought\n<channel|>' in prefix or not prefix.endswith('<|turn>model\n'):
            raise ValueError('SFT must not force an empty thought')
        terminator = '<turn|>\n'
        terminator_ids = tokenizer.encode(terminator, add_special_tokens=False)
        if not terminator_ids or not full.endswith(terminator) or ids[-len(terminator_ids):] != terminator_ids:
            raise ValueError('official assistant turn terminator mismatch')
        frames.append({'id': row['task_id'], 'ids': ids, 'prefix_tokens': len(pids)})
        evidence.append({'id': row['task_id'], 'prefix_tokens': len(pids), 'answer_eos_tokens': len(ids) - len(pids), 'complete_tokens': len(ids), 'prefix_sha256': hashlib.sha256(prefix.encode()).hexdigest(), 'full_sha256': hashlib.sha256(full.encode()).hexdigest(), 'token_ids_sha256': hashlib.sha256(json.dumps(ids).encode()).hexdigest(), 'eligible': len(ids) <= 1024})
    return (frames, {'schema_version': 1, 'frame_protocol': FRAME_PROTOCOL, 'template_sha256': hashlib.sha256(str(tokenizer.chat_template).encode()).hexdigest(), 'enable_thinking': True, 'forced_empty_thought': False, 'padding': False, 'loss': 'mean answer and official turn-terminator tokens; prompt masked; each example equally weighted', 'official_terminator': '<turn|>\n', 'manual_global_eos_added': False, 'maximum_tokens': 1024, 'truncation': False, 'rows': evidence, 'ineligible_ids': [row['id'] for row in evidence if not row['eligible']]})
