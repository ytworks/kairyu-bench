def sha(path):
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 2**20), b""):
            digest.update(block)
    return digest.hexdigest()


def stamp(path):
    s = path.lstat()
    assert stat.S_ISREG(s.st_mode) and not path.is_symlink()
    return dict(device=s.st_dev, inode=s.st_ino, size=s.st_size, mtime_ns=s.st_mtime_ns,
                ctime_ns=s.st_ctime_ns, mode=s.st_mode, uid=s.st_uid, gid=s.st_gid, nlink=s.st_nlink)


def inventory(root):
    assert root.resolve() == root and root.is_dir() and not root.is_symlink()
    paths = sorted(root.rglob("*"))
    assert not any(p.is_symlink() for p in paths)
    return {str(p.relative_to(root)): stamp(p) for p in paths if p.is_file()}


def mount_snapshot():
    ids = subprocess.check_output(["docker", "ps", "-aq", "--no-trunc"], text=True).split()
    data = json.loads(subprocess.check_output(["docker", "inspect", *ids], text=True))
    snapshots, conflicts = [], []
    for d in data:
        mounts = []
        for m in d.get("Mounts", []):
            path = Path(m["Source"]).resolve() if m.get("Source", "").startswith("/") else None
            related = path is not None and (path == F02 or path.is_relative_to(F02) or F02.is_relative_to(path))
            # Save only the selected filesystem mount fields, never environment/config.
            item = {k: m.get(k) for k in ("Type", "Source", "Destination", "RW")}
            item["candidate_overlap"] = related
            mounts.append(item)
            if related:
                conflicts.append({"cid": d["Id"], "name": d["Name"], "mount": item})
        snapshots.append({"cid": d["Id"], "name": d["Name"], "image": d["Image"],
                          "running": d["State"]["Running"], "status": d["State"]["Status"],
                          "health": d["State"].get("Health", {}).get("Status"), "mounts": mounts})
    return sorted(snapshots, key=lambda x: x["cid"]), conflicts

    assert base_before == expected_base and candidate_before == expected_candidate

    assert AV.stat().st_uid == F02.stat().st_uid == 1003
    result.update(base_current_stat=base_before, candidate_current_stat=candidate_before,
                  base9_stat_continuity=True, candidate9_stat_continuity=True)

    for line in (A / "SHA256SUMS").read_text().splitlines():

    result.update(archive_checksum_list_sha256=SUMS, archive_json_sha256=ARCHIVE,
